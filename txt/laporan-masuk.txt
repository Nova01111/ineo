    // Validasi tipe
    if (!file.type.startsWith("image/")) {
      showToast("Format tidak didukung", "error", "Pilih file gambar (JPG, PNG, WEBP)");
      return;
    }
    // Validasi ukuran (maks 5MB)
    if (file.size > 5 * 1024 * 1024) {
      showToast("File terlalu besar", "error", "Ukuran maksimal 5MB");
      return;
    }

    fotoFile        = file;
    fotoUploadedUrl = null; // reset URL lama

    // Tampilkan preview lokal langsung (sebelum upload)
    const reader = new FileReader();
    reader.onload = (e) => {
      document.getElementById("preview-img").src = e.target.result;
    };
    reader.readAsDataURL(file);

    // Info file
    document.getElementById("preview-name").textContent = file.name;
    document.getElementById("preview-size").textContent = formatBytes(file.size);

    // Tampilkan preview, sembunyikan drop zone
    document.getElementById("drop-zone").style.display    = "none";
    document.getElementById("foto-preview").classList.add("show");
    document.getElementById("upload-progress").classList.remove("show");
  }

  // ── Hapus foto yang dipilih ─────────────────────────────────────────────
  function hapusFoto() {
    fotoFile        = null;

    document.getElementById("file-input").value       = "";

  }

  // ── Upload foto ke server, return URL path ──────────────────────────────
  async function uploadFoto(file) {
    const prog      = document.getElementById("upload-progress");
    const bar       = document.getElementById("progress-bar");
    const pct       = document.getElementById("progress-pct");
    const lbl       = document.getElementById("progress-label");

    prog.classList.add("show");
    bar.style.width = "0%";
    pct.textContent = "0%";
    lbl.textContent = "Mengupload...";

    return new Promise((resolve, reject) => {
      const formData = new FormData();
      formData.append("file", file);
      formData.append("laporan_id", laporanAktif);

      const xhr = new XMLHttpRequest();

      // Progress
      xhr.upload.addEventListener("progress", (e) => {
        if (e.lengthComputable) {
          const p = Math.round((e.loaded / e.total) * 100);
          bar.style.width = p + "%";
          pct.textContent = p + "%";
        }
      });

      xhr.addEventListener("load", () => {
        if (xhr.status >= 200 && xhr.status < 300) {
          try {
            const res = JSON.parse(xhr.responseText);
            if (res.success && res.data?.path) {
              bar.style.width = "100%";
              pct.textContent = "100%";
              lbl.textContent = "Upload selesai ✓";
              resolve(res.data.path);
            } else {
              reject(new Error(res.message || "Respons upload tidak valid"));
            }
          } catch {
            reject(new Error("Gagal memproses respons server"));
          }
        } else {
          reject(new Error(`Upload gagal (HTTP ${xhr.status})`));
        }
      });

      xhr.addEventListener("error", () => reject(new Error("Koneksi terputus saat upload")));
      xhr.addEventListener("abort", () => reject(new Error("Upload dibatalkan")));

      // Ambil token auth dari apiFetch (gunakan header yang sama)
      const token = localStorage.getItem("auth_token") || sessionStorage.getItem("auth_token") || "";
      xhr.open("POST", "/admin-pusat/upload/bukti-transfer");
      if (token) xhr.setRequestHeader("Authorization", `Bearer ${token}`);
      xhr.send(formData);
    });
  }

  // ── Format bytes helper ─────────────────────────────────────────────────
  function formatBytes(bytes) {
    if (bytes < 1024)       return bytes + " B";
    if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + " KB";
    return (bytes / (1024 * 1024)).toFixed(1) + " MB";
  }

  // ── Simpan rekap gaji + kirim notifikasi ke admin cabang ────────────────
  async function simpanRekap() {
    const errEl = document.getElementById("error-rekap");
    errEl.style.display = "none";

    const tgl_transfer = document.getElementById("tgl-transfer").value;

    // Tentukan screenshot_path: prioritas foto upload, fallback URL manual
    const tabAktifUpload = document.getElementById("panel-upload-foto").classList.contains("show") ? "foto" : "url";
    let screenshot_path  = "";

    const btn = document.getElementById("btn-simpan-rekap");
    btn.disabled = true;

    // ── LANGKAH 0: Upload foto jika ada file yang dipilih ─────────────────
    if (tabAktifUpload === "foto" && fotoFile) {
      btn.innerHTML = '<span class="spin-sm"></span> Mengupload foto...';
      try {
        fotoUploadedUrl = await uploadFoto(fotoFile);
        screenshot_path = fotoUploadedUrl;
      } catch (err) {
        btn.disabled = false;
        btn.innerHTML = "Simpan & Tandai Dibayar";
        errEl.textContent = `Gagal upload foto: ${err.message}`;
        errEl.style.display = "block";
        return;
      }
    } else if (tabAktifUpload === "foto" && fotoUploadedUrl) {
      // File sudah pernah diupload sebelumnya (tidak perlu re-upload)
      screenshot_path = fotoUploadedUrl;
    } else {
      // Gunakan URL manual
      screenshot_path = document.getElementById("screenshot-url").value.trim();
    }

    btn.innerHTML = '<span class="spin-sm"></span> Menyimpan...';

    // Kumpulkan potongan
    const potongan = [];
    document.querySelectorAll(".pot-row").forEach(row => {
      const nominal = parseFloat(row.querySelector(".pot-nominal").value) || 0;
      if (nominal > 0) {
        potongan.push({
          jenis_potongan_id: parseInt(row.querySelector(".pot-jenis").value),
          keterangan: row.querySelector(".pot-ket").value.trim() || null,
          nominal,
        });
      }
    });

    // Hitung total potongan & gaji bersih untuk payload notifikasi
    const totalPotongan = potongan.reduce((s, p) => s + p.nominal, 0);
    const gajiBersih    = Math.max(0, totalGajiAktif - totalPotongan);

    // ── LANGKAH 1: Simpan rekap gaji ──────────────────────────────────────
    const resRekap = await apiFetch("/admin-pusat/rekap-gaji", {
      method: "POST",
      body: JSON.stringify({
        laporan_wo_id: laporanAktif,
        potongan,
        tgl_transfer:    tgl_transfer    || null,
        screenshot_path: screenshot_path || null,
      }),
    });

    if (!resRekap?.success) {
      btn.disabled = false;
      btn.innerHTML = "Simpan & Tandai Dibayar";
      errEl.textContent = resRekap?.message || "Gagal menyimpan rekap gaji";
      errEl.style.display = "block";
      return;
    }

    // ── LANGKAH 2: Kirim notifikasi ke admin cabang ───────────────────────
    const laporan       = rekapDataAktif?.laporan || {};
    const rekap_gaji_id = resRekap?.data?.rekap_gaji_id || null;

    const payloadNotif = {
      laporan_wo_id:   laporanAktif,
      rekap_gaji_id,
      cabang_id:       laporan.cabang_id,
      nama_teknisi:    laporan.nama_teknisi,
      periode:         `${BLN[laporan.bulan]} ${laporan.tahun}`,
      total_gaji:      totalGajiAktif,
      total_potongan:  totalPotongan,
      gaji_bersih:     gajiBersih,
      tgl_transfer:    tgl_transfer    || null,
      screenshot_path: screenshot_path || null,
    };

    const resNotif = await apiFetch("/admin-pusat/notifikasi/kirim", {
      method: "POST",
      body: JSON.stringify(payloadNotif),
    });

    btn.disabled = false;
    btn.innerHTML = "Simpan & Tandai Dibayar";

    if (resNotif?.success) {
      showToast(
        "Rekap disimpan & notifikasi terkirim",
        "success",
        `Admin cabang ${escHtml(laporan.nama_cabang || "")} telah diberitahu`
      );
    } else {
      showToast(
        "Rekap disimpan",
        "info",
        `Notifikasi gagal dikirim: ${resNotif?.message || "coba refresh halaman admin cabang"}`
      );
    }

    tutupPanel();
    loadLaporan();
  }

  // ── Panel control ───────────────────────────────────────────────────────
  function tutupPanel(e) {
    if (e && e.target !== document.getElementById("detail-overlay")) return;
    document.getElementById("detail-overlay").classList.remove("show");
    document.body.style.overflow = "";
    // Reset state foto
    fotoFile        = null;
    fotoUploadedUrl = null;
  }

  document.addEventListener("keydown", e => {
    if (e.key === "Escape") tutupPanel({ target: document.getElementById("detail-overlay") });
  });

  // Default tampilkan laporan berstatus "terkirim" (perlu verifikasi)

  Promise.all([loadCabang(), loadJenisPotongan()]).then(() => loadLaporan());
